package main

import (
	"fmt"
	"reflect"
	"strings"
)

func main() {
	fmt.Println(strings.Repeat(">", 10))
	str := "hello,world,go"
	// 1. len(str) 求长度（注意：按字节数，不是字符数）
	fmt.Println("len:", len(str)) // 13
	// 2. 字符串拼接
	a := "hello"
	b := "golang"
	fmt.Println("拼接 + :", a+" "+b)
	fmt.Println("拼接 Sprintf:", fmt.Sprintf("%s %s", a, b))
	// 3. strings.Split 分割
	parts := strings.Split(str, ",")
	fmt.Println("Split:", parts)                        // ["hello" "world" "go"]
	fmt.Println("Splited type:", reflect.TypeOf(parts)) // ["hello" "world" "go"]
	// 4. strings.Contains 判断是否包含
	fmt.Println("Contains 'world':", strings.Contains(str, "world")) // true
	// 5. strings.HasPrefix / HasSuffix 前缀/后缀判断
	fmt.Println("HasPrefix 'hello':", strings.HasPrefix(str, "hello")) // true
	fmt.Println("HasSuffix 'go':", strings.HasSuffix(str, "go"))       // true
	// 6. strings.Index / LastIndex 子串位置
	fmt.Println("Index of 'o':", strings.Index(str, "o"))         // 第一次出现的位置
	fmt.Println("LastIndex of 'o':", strings.LastIndex(str, "o")) // 最后一次出现的位置
	// 7. strings.Join 连接 slice
	joined := strings.Join(parts, " | ")
	fmt.Println("Join:", joined) // "hello | world | go"
}

package main

import (
	"fmt"
	"strings"
)

func main() {
	fmt.Println(strings.Repeat(">", 10))
	strr1 := "阿把从的额发个abcdafadfadsfasdfasd"
	strr2 := strr1[0:1]
	strc1 := []rune(strr1)
	strc2 := strc1[0:1]
	fmt.Println("strr1:", string(strr2), "strc1:", string(strc2))
	fmt.Println("strr1:", strr2, "strc1:", strc2)
}
>>>>>>>>>>
strr1: � strc1: 阿
strr1: � strc1: [38463]

输出解析

  1. strr2 := strr1[0:1]
    • strr1 是 UTF-8 字符串。
    • “阿” 的 UTF-8 编码是 3 个字节：E9 98 BF（十六进制）。
    • 你只切了 [0:1] → 只拿了第一个字节 0xE9。
    • 把这个单字节解释成 UTF-8 字符 → 无效字节 → 打印成 � (replacement character)。
  2. strc1 := []rune(strr1)
    • []rune 会按 Unicode 码点解码整个字符串。
    • “阿” 的码点是 U+963F = 38463（十进制）。
    • 所以 strc1[0] = 38463。
    • strc2 := strc1[0:1] → []rune{38463}。
    • 打印时：
      • string(strc2) → 把 rune 38463 转成 UTF-8 → “阿”。
      • fmt.Println(strc2) → 打印 slice → [38463]。

package main

import (
"fmt"
"strings"
)

func main() {
   fmt.Println(strings.Repeat(">", 10))
   // +
   a1 := "hello"
   b1 := "world"
   c1 := a1 + "1" + b1
   fmt.Println("c1", c1)
   // sprintf
   a2 := "hello"
   b2 := "world"
   c2 := fmt.Sprint(a2, b2)
   fmt.Println("c2", c2)
   // join
   a3 := "hello"
   b3 := "world"
   s3 := []string{a3, b3}
   c3 := strings.Join(s3, "3")
   fmt.Println("s3", s3)
   fmt.Println("c3", c3)
   // build.Builder  效率很高
   a4 := "hello"
   b4 := "world"
   var build strings.Builder
   build.WriteString(a4)
   build.WriteString("4")
   build.WriteString(b4)
   c4 := build.String()
   fmt.Println("c4", c4)
}

package main

import (
   "fmt"
   "strconv"
   "strings"
)

func main() {
   fmt.Println(strings.Repeat(">", 10))
   //int转string
   a1 := 65
   b1 := strconv.Itoa(a1)
   c1, _ := strconv.Atoi(b1)
   d1 := string(a1) //通过 ASCII 码转换
   fmt.Printf("a1 value:%v | a1 type:%T | b1 value:%v | b1 type:%T | c1 value:%v ｜ c1 type:%T | d1 value:%v ｜ d1 type:%T\n", a1, a1, b1, b1, c1, c1, d1, d1)
   //string转int
   a2 := "abc"
   b2, err := strconv.Atoi(a2)
   if err != nil {
      fmt.Println("b2 in err", b2)
   }
   fmt.Printf("a2 value:%v | a2 type:%T | b2 value:%v | b2 type:%T \n", a2, a2, b2, b2)
   a3 := "100"
   b3, err := strconv.Atoi(a3)
   if err != nil {
      fmt.Println(b3)
   }
   fmt.Printf("a3 value:%v | a3 type:%T | b3 value:%v | b3 type:%T \n", a3, a3, b3, b3)
}

float和string的相互转换

float转string

语法 

1

str := strconv.FormatFloat(f float64, fmt byte, prec int, bitSize int)

参数

参数 描述
str 将 float 转换成的字符串。
f 需要转换的 float64 类型的变量。
fmt 使用 f 表示不使用指数的形式。
prec 保留几位小数。
bitSize 如果为 32，表示是 float32 类型，如果是 64，表示是 float64 类型。

说明

我们使用了 strconv.FormatFloat 实现了将 float 类型的变量 f，转换了字符串类型的变量 str。

注意

strconv.FormatFloat 函数的第一个参数，只能接受 float64 类型的变量。因此，如果需要将 float32 类型转成 string，需要先将 float32 转成 float64。

字符串的比较

Go 语言 中比较两个 字符串 是否相等有三种方法，分别是：使用 == 比较、使用 Compare 比较和使用 EqualFold 比较。

Go语言字符串等于比较

语法 

1

str1 == str2

返回值

如果相等，则返回 true，否则，返回 false。

说明

比较字符串 str1 和字符串 str2 是否相等。

字符串Compare比较

语法 

1

func Compare(a, b string) int

参数

参数 描述
a 需要比较的字符串。
b 需要比较的字符串。

返回值

比较字符串 a 和字符串 b 是否相等，如果 a > b，返回一个大于 0 的数，如果 a == b，返回 0，否则，返回负数。
