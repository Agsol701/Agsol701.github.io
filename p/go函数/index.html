package main

import (
	"fmt"
	"strings"
)

func changebyptr(x *int) {
	*x += 100
}

func changebycal(x int) (r int) {
	r = x + 100
	return r
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	a := 100
	fmt.Println("a", a)
	changebyptr(&a)
	fmt.Println("a", a)
	r := changebycal(a)
	fmt.Println("r", r)
	fmt.Println("a", a)
}

package main

import (
	"fmt"
	"strings"
)

func manytypePrint(args ...interface{}) {
	for _, arg := range args {
		switch arg.(type) {
		case int:
			fmt.Println("int")
		case string:
			fmt.Println("string")
		}
	}
}

func onetypePrint(args interface{}) {
	switch args.(type) {
	case int:
		fmt.Println("int")
	case string:
		fmt.Println("string")
	}
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	a := "abc"
	b := 8848
	manytypePrint(a, b)
	onetypePrint(a)
	onetypePrint(b)
}

package main

import (
	"fmt"
	"strings"
)

func adder() func(int) int {
	var x int
	return func(i int) int {
		x += i
		return x
	}
}

func adder2(x int) func(int) int {
	return func(y int) int {
		x += y
		return x
	}
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	var f = adder()
	fmt.Println(f(10))
	fmt.Println(f(20))
	fmt.Println(f(30))
	fmt.Println(strings.Repeat("-", 10))
	f1 := adder()
	fmt.Println(f1(40))
	fmt.Println(f1(50))
	fmt.Println(f1(60))
	fmt.Println(strings.Repeat("-", 20))
	var f2 = adder2(10)
	fmt.Println(f2(10)) //20
	fmt.Println(f2(20)) //40
	fmt.Println(f2(30)) //70
	fmt.Println(strings.Repeat("-", 10))
	f2_1 := adder2(20)
	fmt.Println(f2_1(40)) //60
	fmt.Println(f2_1(50)) //110
}

见到defer就压栈，外层return后出栈执行 Go 语言 的 defer 语句 一般都是用来处理需要关闭的资源。

defer、return、返回值的执行返回值顺序

在此之前，先理解一下return返回值的运行机制：return并非原子操作，共分为(1) 返回值赋值；(2) RET指令两步操作。而defer语句执行在赋值之后，RET之前。 defer、return、返回值三者的执行是：return最先执行，先将结果写入返回值中（即赋值）；接着defer开始执行一些收尾工作；最后函数携带当前返回值退出（即返回值）。

package main

import (
	"errors"
	"fmt"
	"strings"
)

var diverr = errors.New("division by zero")

func div(a, b int) (res int, err error) {
	if b == 0 {
		return 0, diverr
	}
	return a / b, nil
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	res1, err := div(10, 2)
	if err == nil {
		fmt.Println(res1)
	}
	res2, err := div(10, 0)
	if err != nil {
		fmt.Println(err)
	}
	fmt.Println(res2)
}

recover

recover只有在defer调用的函数中有效。

panic和recover使用原则

  • defer 需要放在 panic 之前定义，另外 recover 只有在 defer 调用的函数中才有效。
  • recover 处理异常后，逻辑并不会恢复到 panic 那个点去，函数跑到 defer 之后的那个点。

Go语言panic和recover的关系

如果有 panic 但没有 recover，那么程序会宕机。如果有 panic 也有 recover，程序不会宕机，执行完对应的 defer 后，从宕机点退出当前函数后继续执行。 虽然 panic/recover 能模拟其他语言的异常机制，但并不建议在编写普通函数时也经常性使用这种特性。在 panic 触发的 defer 函数内，可以继续调用 panic，进一步将错误外抛，直到程序整体崩溃。

package main

import (
	"fmt"
	"strings"
)

func main() {
	fmt.Println(strings.Repeat(">", 20))
	defer func() {
		if info := recover(); info != nil {
			fmt.Println("宕机", info)
		} else {
			fmt.Println("正常退出程序")
		}
	}()
	fmt.Println("1")
	panic("panic")
	fmt.Println("2")
}

	fmt.Println("func C")
}

	fmt.Println(strings.Repeat(">", 20))
	funcA()
	funcB()

package main

import (
	"fmt"
	"strings"
)

func add(x, y int) int {
	return x + y
}

func minus(x, y int) int {
	return x - y
}

func callBack(s string) func(int, int) int {
	switch s {
	case "+":
		return add
	case "-":
		return minus
	default:
		return nil
	}
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	addRes := callBack("+")(1, 4)
	fmt.Println(addRes)
	minusRes := callBack("-")(5, 3)
	fmt.Println(minusRes)
}
