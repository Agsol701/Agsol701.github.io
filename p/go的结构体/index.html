Oct 06, 2025

结构体的定义

使用type和struct关键字来定义结构体，具体代码格式如下：

1
2
3
4
5

type 类型名 struct {
    字段名 字段类型
    字段名 字段类型
	...
}

其中：

  • 类型名：标识自定义结构体的名称，在同一个包内不能重复。
  • 字段名：表示结构体字段名。结构体中的字段名必须唯一。
  • 字段类型：表示结构体字段的具体类型。

结构体的初始化

没有初始化的结构体，其成员变量都是对应其类型的零值。

接收器receiver

为结构体添加方法

就是python面向对象中的类的方法

说明

  • Go 语言中我们除了可以单独定义函数，同时，还可以为结构体添加函数。被添加方法的结构体的变量名在 Golang 中被叫做接收器。
  • 这里的receiver，我们在 Golang 中叫做方法的接收器，Golang 中的方法接收器在命名时，官方建议使用接收器类型名的第一个小写字母。
  • Golang 中接收器类型可以是任何类型，比如 int、bool、string、数组、切片等类型的别名，甚至也可以是函数类型。 为结构体添加方法语法：
1
2

var (s StructType)funcName(param1 param1Type, ...)(returnVal1 returnType1, ...{
}
  • Go 语言中结构体方法的接收器可以是一个指针类型，如果是指针类型的接收器，那么在方法中，对结构体的修改都会生效。为结构体添加方法指针接收器语法：
1
2

var (s *StructType)funcName(param1 param1Type, ...)(returnVal1 returnType1, ...){
}

package main

import (
	"fmt"
	"strings"
)

// Person 结构体
type Person struct {
	name string
	age  int8
}

// NewPerson 构造函数
func NewPerson(name string, age int8) *Person {
	return &Person{
		name: name,
		age:  age,
	}
}

// 值类型 receiver
func (p Person) Dream() {
	fmt.Printf("%s的梦想是学好Go语言！\n", p.name)
}

// 指针类型 receiver
func (p *Person) SetAge(newAge int8) {
	p.age = newAge
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	p1 := NewPerson("agsol", 25)
	p1.Dream()
	fmt.Println(*p1)
	fmt.Println(strings.Repeat(">", 10))
	p1.SetAge(50)
	fmt.Println(*p1)
}

值类型和指针类型选型

  1. 需要修改接收者中的值
  2. 接收者是拷贝代价比较大的大对象
  3. 保证一致性，如果有某个方法使用了指针接收者，那么其他的方法也应该使用指针接收者。

	"fmt"
	"strings"
)

type Animal struct {
	Name string
}

func (a Animal) Eat() {
	fmt.Println("animal", a.Name, "is eating")
}

type Dog struct {
	Animal
	Say string
}

func (d Dog) Eat() {
	fmt.Println("dog", d.Name, "is eating")
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	dog := Dog{
		Animal: Animal{Name: "spike"},
		Say:    "woof",
	}
	fmt.Println("dog:", dog)
	fmt.Println("dog name:", dog.Name)
	fmt.Println("dog say:", dog.Say)
	fmt.Println("dog animal name:", dog.Animal.Name)
	dog.Eat()
	dog.Animal.Eat()
}

package main

import (
	"encoding/json"
	"fmt"
	"strings"
)

// Student 学生
type Student struct {
	ID     int
	Gender string
	Name   string
}

// Class 班级
type Class struct {
	Title   string
	Student []Student
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	c := &Class{
		Title:   "101",
		Student: make([]Student, 0, 200),
	}
	for i := range 3 {
		stu := Student{
			ID:     i,
			Gender: "男",
			Name:   fmt.Sprintf("stu%02d", i),
		}
		c.Student = append(c.Student, stu)
	}
	//JSON序列化：结构体-->JSON格式的字符串
	jsonCData, err := json.Marshal(c)
	if err != nil {
		fmt.Println("json marshal failed")
		return
	}
	fmt.Println("json_c_data:", string(jsonCData))
	//JSON序列化：结构体-->JSON格式的字符串
	cDataStr := `{"Title":"101","Student":[{"ID":0,"Gender":"男","Name":"stu00"},{"ID":1,"Gender":"男","Name":"stu01"},{"ID":2,"Gender":"男","Name":"stu02"}]}`
	c1 := &Class{}
	err = json.Unmarshal([]byte(cDataStr), c1)
	if err != nil {
		fmt.Println("json unmarshal failed")
		return
	}
	fmt.Println("c1:", *c1)
}

package main

import (
	"encoding/json"
	"fmt"
	"strings"
)

type Person struct {
	PName string `json:"Name"`
	PAge  int    `json:"Age"`
}

func main() {
	fmt.Println(strings.Repeat(">", 20))
	jsonStr1 := `{"Name":"agsol1","Age":18}`
	jsonStr2 := `{"PName":"agsol2","PAge":28}`
	var p1 Person
	var p2 Person
	p3 := Person{
		PName: "agsol3",
		PAge:  38,
	}
	if err := json.Unmarshal([]byte(jsonStr1), &p1); err != nil {
		fmt.Println(err)
		return
	}
	fmt.Println("p1", p1)
	if err := json.Unmarshal([]byte(jsonStr2), &p2); err != nil {
		fmt.Println(err)
		return
	}
	fmt.Println("p2", p2)	//json中的字段名必须和结构体中的字段名一致，否则会解析失败
	if jsonPerson, err := json.Marshal(p3); err != nil {
		fmt.Println("p3 err:", err)
	} else {
		fmt.Println("p3:", string(jsonPerson))
	}
}

	name   string

func main() {

	agsolDream := []string{"money", "power", "love"}
