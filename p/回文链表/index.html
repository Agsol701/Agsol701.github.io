回文链表

Dec 06, 2025

题目描述

原题链接：https://leetcode.cn/problems/palindrome-linked-list/ 给你一个单链表的头节点 head ，请你判断该链表是否为回文链表。如果是，返回 true ；否则，返回 false 。

示例 1：

1
2

输入：head = [1,2,2,1]
输出：true

示例 2：

1
2

输入：head = [1,2]
输出：false

提示：

  • 链表中节点数目在范围[1, 105] 内
  • 0 <= Node.val <= 9

**进阶：**你能否用 O(n) 时间复杂度和 O(1) 空间复杂度解决此题？

from typing import Optional


# Definition for singly-linked list.
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class Solution:
    def isPalindrome(self, head: Optional[ListNode]) -> bool:
        if not head or not head.next:
            return True

        slow = fast = head
        # 这两个条件都需要存在，因为无法保证链表是奇数个或偶数个，防止越界
        while fast and fast.next:
            fast = fast.next.next
            slow = slow.next

        # 构建逆转的后半部分链表
        prev = None
        cur = slow
        while cur:
            nxt = cur.next
            cur.next = prev
            prev = cur
            cur = nxt

        # 判断是否为回文
        while prev:
            if prev.val != head.val:
                return False
            prev = prev.next
            head = head.next

        return True


# 工具函数：列表 -> 链表
def build_linked_list(arr):
    if not arr:
        return None
    head = ListNode(arr[0])
    cur = head
    for x in arr[1:]:
        cur.next = ListNode(x)
        cur = cur.next
    return head


# 工具函数：打印链表（调试用）
def print_list(head):
    arr = []
    while head:
        arr.append(head.val)
        head = head.next
    print(arr)


head = build_linked_list([1, 2])
print(Solution().isPalindrome(head))

package main

// Definition for singly-linked list.
type ListNode struct {
	Val  int
	Next *ListNode
}

func isPalindrome(head *ListNode) bool {
	if head == nil || head.Next == nil {
		return true
	}
	fast := head
	slow := head
	for fast != nil && fast.Next != nil {
		slow = slow.Next
		fast = fast.Next.Next
	}
	var prev *ListNode = nil
	cur := slow
	for cur != nil {
		nxt := cur.Next
		cur.Next = prev
		prev = cur
		cur = nxt
	}
	for prev != nil {
		if prev.Val != head.Val {
			return false
		}
		prev = prev.Next
		head = head.Next
	}
	return true
}
