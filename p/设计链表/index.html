设计链表

Apr 13, 2026

题目描述

原题链接：https://leetcode.cn/problems/design-linked-list/

设计链表的实现。你可以选择使用单链表或者双链表。单链表中的节点应该具有两个属性：val 和 next。val 是当前节点的值，next 是指向下一个节点的指针/引用。

假设链表中的所有节点都是 0-index 的。

在链表类中实现这些功能：

  • get(index)：获取链表中第 index 个节点的值。如果索引无效，则返回 -1。
  • addAtHead(val)：在链表的第一个元素之前添加一个值为 val 的节点。插入后，新节点将成为链表的第一个节点。
  • addAtTail(val)：将值为 val 的节点追加到链表的最后一个元素。
  • addAtIndex(index,val)：在链表中的第 index 个节点之前添加值为 val 的节点。如果 index 等于链表的长度，则该节点将附加到链表的末尾。如果 index 大于链表长度，则不会插入节点。如果 index 小于 0，则在头部插入节点。
  • deleteAtIndex(index)：如果索引 index 有效，则删除链表中的第 index 个节点。

MyLinkedList linkedList = new MyLinkedList();

提示：

  • 所有 val 值都在 [1, 1000] 之内。
  • 操作次数将在 [1, 1000] 之内。
  • 请不要使用内置的 LinkedList 库。

解题思路

这道题需要实现一个完整的链表类，支持增删查操作。可以采用单链表或双链表两种方式实现。

单链表

使用虚拟头节点 dummy 可以统一处理头节点的插入和删除，避免特殊判断。同时维护一个 size 变量记录链表长度，方便边界检查。

各操作的实现思路：

  1. get(index)：从 dummy.next 开始遍历 index 步，返回对应节点的值。
  2. addAtHead(val)：在 dummy 后面插入新节点。
  3. addAtTail(val)：遍历到最后一个节点，在其后面插入新节点。
  4. addAtIndex(index, val)：找到第 index - 1 个节点，在其后面插入新节点。
  5. deleteAtIndex(index)：找到第 index - 1 个节点，删除其后继节点。

时间复杂度：get、addAtTail、addAtIndex、deleteAtIndex 都是 O(n)，addAtHead 是 O(1)

双链表

双链表节点包含 prev 和 next 两个指针，可以双向遍历。使用虚拟头节点 dummyHead 和虚拟尾节点 dummyTail，可以统一处理头尾的插入和删除。

各操作的实现思路：

  1. get(index)：根据 index 与 size/2 的关系，决定从头还是从尾遍历，提高效率。
  2. addAtHead(val)：在 dummyHead 后面插入新节点。
  3. addAtTail(val)：在 dummyTail 前面插入新节点。
  4. addAtIndex(index, val)：找到第 index 个节点，在其前面插入新节点。
  5. deleteAtIndex(index)：找到第 index 个节点，删除它。

时间复杂度：get 最优 O(n/2)，addAtHead 和 addAtTail 都是 O(1)

单链表（不使用 size）

不维护 size 变量，每次操作时通过遍历来确定链表长度或判断索引是否越界。

各操作的实现思路：

  1. get(index)：从 dummy.next 开始遍历 index 步，如果遍历过程中遇到 None，说明索引越界，返回 -1。
  2. addAtHead(val)：直接在 dummy 后面插入新节点。
  3. addAtTail(val)：遍历到最后一个节点（next 为 None），在其后面插入新节点。
  4. addAtIndex(index, val)：遍历找第 index 个节点的前一个位置，如果遍历过程中提前遇到 None，说明越界。
  5. deleteAtIndex(index)：遍历找第 index 个节点的前一个位置，如果其后继为 None，说明越界。

注意：不使用 size 时，判断越界需要在遍历过程中检查节点是否为 None。

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class MyLinkedList:

    def __init__(self):
        # dummy 是虚拟头节点，始终在 head 前一个位置
        # 初始化时链表为空，dummy.next = None
        # 第一次 addAtHead(1) 后：dummy -> 1 -> None
        # 无论链表是否为空，dummy 始终是 head 的前一个节点
        self.dummy = ListNode(0)
        self.size = 0

    def get(self, index: int) -> int:
        if index < 0 or index >= self.size:
            return -1
        cur = self.dummy.next
        for _ in range(index):
            cur = cur.next
        return cur.val

    def addAtHead(self, val: int) -> None:
        # 在 dummy 后面插入新节点，新节点成为新的 head
        # dummy 始终是 head 的前一个节点
        new_node = ListNode(val, self.dummy.next)
        self.dummy.next = new_node
        self.size += 1

    def addAtTail(self, val: int) -> None:
        cur = self.dummy
        while cur.next:
            cur = cur.next
        cur.next = ListNode(val)
        self.size += 1

    def addAtIndex(self, index: int, val: int) -> None:
        if index > self.size:
            return
        if index < 0:
            index = 0
        prev = self.dummy
        for _ in range(index):
            prev = prev.next
        prev.next = ListNode(val, prev.next)
        self.size += 1

    def deleteAtIndex(self, index: int) -> None:
        if index < 0 or index >= self.size:
            return
        prev = self.dummy
        for _ in range(index):
            prev = prev.next
        prev.next = prev.next.next
        self.size -= 1

type ListNode struct {
	Val  int
	Next *ListNode
}

type MyLinkedList struct {
	dummy *ListNode
	size  int
}

func Constructor() MyLinkedList {
	// dummy 是虚拟头节点，始终在 head 前一个位置
	// 初始化时链表为空，dummy.Next = nil
	// 第一次 AddAtHead(1) 后：dummy -> 1 -> nil
	// 无论链表是否为空，dummy 始终是 head 的前一个节点
	return MyLinkedList{
		dummy: &ListNode{},
		size:  0,
	}
}

func (this *MyLinkedList) Get(index int) int {
	if index < 0 || index >= this.size {
		return -1
	}
	cur := this.dummy.Next
	for i := 0; i < index; i++ {
		cur = cur.Next
	}
	return cur.Val
}

func (this *MyLinkedList) AddAtHead(val int) {
	// 在 dummy 后面插入新节点，新节点成为新的 head
	// dummy 始终是 head 的前一个节点
	newNode := &ListNode{Val: val, Next: this.dummy.Next}
	this.dummy.Next = newNode
	this.size++
}

func (this *MyLinkedList) AddAtTail(val int) {
	cur := this.dummy
	for cur.Next != nil {
		cur = cur.Next
	}
	cur.Next = &ListNode{Val: val}
	this.size++
}

func (this *MyLinkedList) AddAtIndex(index int, val int) {
	if index > this.size {
		return
	}
	if index < 0 {
		index = 0
	}
	prev := this.dummy
	for i := 0; i < index; i++ {
		prev = prev.Next
	}
	prev.Next = &ListNode{Val: val, Next: prev.Next}
	this.size++
}

func (this *MyLinkedList) DeleteAtIndex(index int) {
	if index < 0 || index >= this.size {
		return
	}
	prev := this.dummy
	for i := 0; i < index; i++ {
		prev = prev.Next
	}
	prev.Next = prev.Next.Next
	this.size--
}

class ListNode:
    def __init__(self, val=0, prev=None, next=None):
        self.val = val
        self.prev = prev
        self.next = next


class MyLinkedList:

    def __init__(self):
        self.dummyHead = ListNode(0)
        self.dummyTail = ListNode(0)
        self.dummyHead.next = self.dummyTail
        self.dummyTail.prev = self.dummyHead
        self.size = 0

    def get(self, index: int) -> int:
        if index < 0 or index >= self.size:
            return -1
        # 根据index位置决定从头还是从尾遍历
        if index < self.size // 2:
            cur = self.dummyHead.next
            for _ in range(index):
                cur = cur.next
        else:
            cur = self.dummyTail.prev
            for _ in range(self.size - index - 1):
                cur = cur.prev
        return cur.val

    def addAtHead(self, val: int) -> None:
        new_node = ListNode(val, self.dummyHead, self.dummyHead.next)
        self.dummyHead.next.prev = new_node
        self.dummyHead.next = new_node
        self.size += 1

    def addAtTail(self, val: int) -> None:
        new_node = ListNode(val, self.dummyTail.prev, self.dummyTail)
        self.dummyTail.prev.next = new_node
        self.dummyTail.prev = new_node
        self.size += 1

    def addAtIndex(self, index: int, val: int) -> None:
        if index > self.size:
            return
        if index < 0:
            index = 0
        # 找到第index个节点
        if index < self.size // 2:
            cur = self.dummyHead.next
            for _ in range(index):
                cur = cur.next
        else:
            cur = self.dummyTail
            for _ in range(self.size - index):
                cur = cur.prev
        # 在cur前面插入新节点
        new_node = ListNode(val, cur.prev, cur)
        cur.prev.next = new_node
        cur.prev = new_node
        self.size += 1

    def deleteAtIndex(self, index: int) -> None:
        if index < 0 or index >= self.size:
            return
        # 找到第index个节点
        if index < self.size // 2:
            cur = self.dummyHead.next
            for _ in range(index):
                cur = cur.next
        else:
            cur = self.dummyTail.prev
            for _ in range(self.size - index - 1):
                cur = cur.prev
        # 删除cur节点
        cur.prev.next = cur.next
        cur.next.prev = cur.prev
        self.size -= 1

type ListNode struct {
	Val  int
	Prev *ListNode
	Next *ListNode
}

type MyLinkedList struct {
	dummyHead *ListNode
	dummyTail *ListNode
	size      int
}

func Constructor() MyLinkedList {
	dummyHead := &ListNode{}
	dummyTail := &ListNode{}
	dummyHead.Next = dummyTail
	dummyTail.Prev = dummyHead
	return MyLinkedList{
		dummyHead: dummyHead,
		dummyTail: dummyTail,
		size:      0,
	}
}

func (this *MyLinkedList) Get(index int) int {
	if index < 0 || index >= this.size {
		return -1
	}
	var cur *ListNode
	if index < this.size/2 {
		cur = this.dummyHead.Next
		for i := 0; i < index; i++ {
			cur = cur.Next
		}
	} else {
		cur = this.dummyTail.Prev
		for i := 0; i < this.size-index-1; i++ {
			cur = cur.Prev
		}
	}
	return cur.Val
}

func (this *MyLinkedList) AddAtHead(val int) {
	newNode := &ListNode{Val: val, Prev: this.dummyHead, Next: this.dummyHead.Next}
	this.dummyHead.Next.Prev = newNode
	this.dummyHead.Next = newNode
	this.size++
}

func (this *MyLinkedList) AddAtTail(val int) {
	newNode := &ListNode{Val: val, Prev: this.dummyTail.Prev, Next: this.dummyTail}
	this.dummyTail.Prev.Next = newNode
	this.dummyTail.Prev = newNode
	this.size++
}

func (this *MyLinkedList) AddAtIndex(index int, val int) {
	if index > this.size {
		return
	}
	if index < 0 {
		index = 0
	}
	var cur *ListNode
	if index < this.size/2 {
		cur = this.dummyHead.Next
		for i := 0; i < index; i++ {
			cur = cur.Next
		}
	} else {
		cur = this.dummyTail
		for i := 0; i < this.size-index; i++ {
			cur = cur.Prev
		}
	}
	newNode := &ListNode{Val: val, Prev: cur.Prev, Next: cur}
	cur.Prev.Next = newNode
	cur.Prev = newNode
	this.size++
}

func (this *MyLinkedList) DeleteAtIndex(index int) {
	if index < 0 || index >= this.size {
		return
	}
	var cur *ListNode
	if index < this.size/2 {
		cur = this.dummyHead.Next
		for i := 0; i < index; i++ {
			cur = cur.Next
		}
	} else {
		cur = this.dummyTail.Prev
		for i := 0; i < this.size-index-1; i++ {
			cur = cur.Prev
		}
	}
	cur.Prev.Next = cur.Next
	cur.Next.Prev = cur.Prev
	this.size--
}

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class MyLinkedList:

    def __init__(self):
        # dummy 是虚拟头节点，始终在 head 前一个位置
        # 初始化时链表为空，dummy.next = None
        # 第一次 addAtHead(1) 后：dummy -> 1 -> None
        # 无论链表是否为空，dummy 始终是 head 的前一个节点
        self.dummy = ListNode(0)

    def get(self, index: int) -> int:
        cur = self.dummy.next
        for _ in range(index):
            if cur is None:
                return -1
            cur = cur.next
        return cur.val if cur else -1

    def addAtHead(self, val: int) -> None:
        # 在 dummy 后面插入新节点，新节点成为新的 head
        # dummy 始终是 head 的前一个节点
        new_node = ListNode(val, self.dummy.next)
        self.dummy.next = new_node

    def addAtTail(self, val: int) -> None:
        cur = self.dummy
        while cur.next:
            cur = cur.next
        cur.next = ListNode(val)

    def addAtIndex(self, index: int, val: int) -> None:
        if index < 0:
            index = 0
        prev = self.dummy
        for _ in range(index):
            if prev.next is None:
                return  # index 超过链表长度，不插入
            prev = prev.next
        prev.next = ListNode(val, prev.next)

    def deleteAtIndex(self, index: int) -> None:
        if index < 0:
            return
        prev = self.dummy
        for _ in range(index):
            if prev.next is None:
                return  # index 越界
            prev = prev.next
        if prev.next:
            prev.next = prev.next.next

type ListNode struct {
	Val  int
	Next *ListNode
}

type MyLinkedList struct {
	dummy *ListNode
}

func Constructor() MyLinkedList {
	// dummy 是虚拟头节点，始终在 head 前一个位置
	// 初始化时链表为空，dummy.Next = nil
	// 第一次 AddAtHead(1) 后：dummy -> 1 -> nil
	// 无论链表是否为空，dummy 始终是 head 的前一个节点
	return MyLinkedList{
		dummy: &ListNode{},
	}
}

func (this *MyLinkedList) Get(index int) int {
	cur := this.dummy.Next
	for i := 0; i < index; i++ {
		if cur == nil {
			return -1
		}
		cur = cur.Next
	}
	if cur == nil {
		return -1
	}
	return cur.Val
}

func (this *MyLinkedList) AddAtHead(val int) {
	// 在 dummy 后面插入新节点，新节点成为新的 head
	// dummy 始终是 head 的前一个节点
	newNode := &ListNode{Val: val, Next: this.dummy.Next}
	this.dummy.Next = newNode
}

func (this *MyLinkedList) AddAtTail(val int) {
	cur := this.dummy
	for cur.Next != nil {
		cur = cur.Next
	}
	cur.Next = &ListNode{Val: val}
}

func (this *MyLinkedList) AddAtIndex(index int, val int) {
	if index < 0 {
		index = 0
	}
	prev := this.dummy
	for i := 0; i < index; i++ {
		if prev.Next == nil {
			return
		}
		prev = prev.Next
	}
	prev.Next = &ListNode{Val: val, Next: prev.Next}
}

func (this *MyLinkedList) DeleteAtIndex(index int) {
	if index < 0 {
		return
	}
	prev := this.dummy
	for i := 0; i < index; i++ {
		if prev.Next == nil {
			return
		}
		prev = prev.Next
	}
	if prev.Next != nil {
		prev.Next = prev.Next.Next
	}
}
