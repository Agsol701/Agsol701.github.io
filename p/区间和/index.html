区间和

Apr 07, 2025

题目描述

题目链接（KamaCoder）：https://kamacoder.com/problempage.php?pid=1070

给定一个整数数组 Array，请计算该数组在每个指定区间内元素的总和。

输入描述：

  • 第一行输入为整数数组 Array 的长度 n
  • 接下来 n 行，每行一个整数，表示数组的元素
  • 随后的输入为需要计算总和的区间下标 a、b（b >= a），直至文件结束

输出描述：

  • 输出每个指定区间内元素的总和

输入示例：

1
2
3
4
5
6
7

1

输出示例：

2

3
9

提示信息：

  • 数据范围：0 < n <= 100000

解题思路

这道题适合使用 前缀和 来优化区间求和。

如果每次遇到一个区间 [a, b] 都重新遍历数组求和，那么一次查询的时间复杂度是 O(b-a+1)。 当查询很多次时，总体效率会比较低。

这类题目的特点是：

  • 原数组内容不会发生修改
  • 需要进行多次区间求和查询

因此非常适合先做一次预处理，再快速回答每次查询。

因此可以先预处理一个前缀和数组 prefix：

  • prefix[i] 表示原数组下标 0 到 i-1 的元素和
  • 也就是 prefix[0] = 0
  • prefix[1] = array[0]
  • prefix[2] = array[0] + array[1]

这里前缀和数组的长度之所以是 n + 1，是因为需要额外保存一个“前 0 个元素的和”，也就是 prefix[0] = 0。

这样做的好处是可以统一使用下面的区间和公式，而不需要对 a = 0 这样的情况单独特判。

这样一来，区间 [a, b] 的和就可以直接用下面的公式得到：

1

sum(a, b) = prefix[b + 1] - prefix[a]

为什么是这个公式？

  • prefix[b + 1] 表示从 0 到 b 的总和
  • prefix[a] 表示从 0 到 a - 1 的总和
  • 两者相减，剩下的正好就是区间 [a, b] 的和

例如数组为 [1, 2, 3, 4, 5]：

1

prefix = [0, 1, 3, 6, 10, 15]

那么：

  • 区间 [0, 1] 的和 = prefix[2] - prefix[0] = 3
  • 区间 [1, 3] 的和 = prefix[4] - prefix[1] = 9

时间复杂度：

  • 预处理前缀和数组：O(n)
  • 每次区间查询：O(1)
  • 总时间复杂度：O(n + q)，其中 q 为查询次数

空间复杂度：O(n)

  • 需要一个长度为 n + 1 的前缀和数组

import sys


def main():
    input = sys.stdin.readline

    n_line = input().strip()
    if not n_line:
        return
    n = int(n_line)

    nums = [0] * n
    for i in range(n):
        nums[i] = int(input().strip())

    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + nums[i]

    res = []
    for line in sys.stdin:
        left, right = map(int, line.strip().split())
        res.append(str(prefix[right + 1] - prefix[left]))

    sys.stdout.write("\n".join(res))


if __name__ == "__main__":
    main()

package main

import (
	"bufio"
	"fmt"
	"os"
)

func main() {
	// 使用带缓冲的输入输出，提高读写效率
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()

	var n int
	// 读取数组长度，如果没有输入则直接结束
	if _, err := fmt.Fscan(in, &n); err != nil {
		return
	}

	// 读取原数组
	nums := make([]int, n)
	for i := 0; i < n; i++ {
		fmt.Fscan(in, &nums[i])
	}

	// prefix[i] 表示 nums[0:i] 的元素和
	prefix := make([]int, n+1)
	for i := 0; i < n; i++ {
		prefix[i+1] = prefix[i] + nums[i]
	}

	var left, right int
	for {
		// 持续读取区间 [left, right]，直到文件结束
		if _, err := fmt.Fscan(in, &left, &right); err != nil {
			break
		}
		// 区间和 = prefix[right + 1] - prefix[left]
		fmt.Fprintln(out, prefix[right+1]-prefix[left])
	}
}

def sum_of_squares_right_add_1(left: int, right: int, ipt_arr: list[int]) -> int:
    prefix = [0] * (len(ipt_arr) + 1)
    for i in range(len(ipt_arr)):
        prefix[i + 1] = prefix[i] + ipt_arr[i]
    return prefix[right + 1] - prefix[left]


def main():
    left = 0
    right = 0
    ipt_arr = [5]
    print(sum_of_squares_right_add_1(left, right, ipt_arr))

package main

import "fmt"

func sumOfSquaresRightAdd1(left, right int, ipt_arr []int) int {
	prefix := make([]int, len(ipt_arr)+1)
	for i := 0; i < len(ipt_arr); i++ {
		prefix[i+1] = prefix[i] + ipt_arr[i]
	}
	return prefix[right+1] - prefix[left]
}

func main() {
	left := 0
	right := 0
	ipt_arr := []int{5}
	fmt.Println(sumOfSquaresRightAdd1(left, right, ipt_arr))
}
