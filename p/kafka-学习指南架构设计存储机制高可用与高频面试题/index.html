# 消费者服务 (Python)
from kafka import KafkaConsumer
import json
import os
from datetime import datetime

class LogConsumer:
    def __init__(self):
        self.consumer = KafkaConsumer(
            'app-logs',
            bootstrap_servers=['kafka1:9092', 'kafka2:9092', 'kafka3:9092'],
            group_id='log-consumer-group',
            auto_offset_reset='earliest',
            enable_auto_commit=True,
            value_deserializer=lambda m: json.loads(m.decode('utf-8')),
            consumer_timeout_ms=10000
        )
        
        # 落盘目录
        self.base_path = '/data/logs'
        
        # 按日期分目录
        self.today = datetime.now().strftime('%Y%m%d')
        
        # 按服务分文件
        self.file_handles = {}
    
    def get_file_handle(self, service_name):
        """获取或创建文件句柄"""
        if service_name not in self.file_handles:
            # 检查日期是否变化
            current_date = datetime.now().strftime('%Y%m%d')
            if current_date != self.today:
                self.today = current_date
                self.close_all_files()
            
            # 创建目录
            dir_path = os.path.join(self.base_path, self.today)
            os.makedirs(dir_path, exist_ok=True)
            
            # 创建文件
            file_path = os.path.join(dir_path, f'{service_name}.log')
            self.file_handles[service_name] = open(file_path, 'a', buffering=1)
        
        return self.file_handles[service_name]
    
    def consume(self):
        """消费消息并落盘"""
        print("开始消费日志...")
        
        for message in self.consumer:
            try:
                log_data = message.value
                
                # 提取服务名
                service_name = log_data.get('service', 'app')
                
                # 写入文件
                file_handle = self.get_file_handle(service_name)
                log_line = json.dumps(log_data) + '\n'
                file_handle.write(log_line)
                
                # 定期 flush (可选)
                # file_handle.flush()
                
            except Exception as e:
                print(f"处理消息失败: {e}")
    
    def close_all_files(self):
        """关闭所有文件句柄"""
        for fh in self.file_handles.values():
            fh.close()
        self.file_handles = {}

# 启动消费者
if __name__ == '__main__':
    consumer = LogConsumer()
    try:
        consumer.consume()
    except KeyboardInterrupt:
        consumer.close_all_files()

│     - 消费者故障恢复后可以从上次位置继续                    │
│                                                             │
│  4. 削峰填谷                                                │
│     - 日志产生不均匀 (高峰期/低谷)                          │
│     - Kafka 缓冲，消费者按自身速度消费                      │
│                                                             │

│     - 可以对接多种下游: ES/HDFS/Hive 等                    │

from kafka import KafkaProducer
from kafka.producer.partitioner import Partitioner
from kafka.errors import kafka_errors
import mmh3  # murmur2 hash 库

class CustomPartitioner(Partitioner):
    
    def __init__(self, partitions=None, **kwargs):
        super().__init__(partitions, **kwargs)
        # 从配置中读取特殊分区号
        self.special_partition = int(kwargs.get('special.partition', 0))
    
    def partition(self, topic, key, key_bytes, value, value_bytes, cluster):
        # 获取主题的分区数
        num_partitions = cluster.partition_count_for_topic(topic)
        
        if key is not None:
            key_str = str(key)
            
            # 自定义逻辑示例：
            # 1. VIP 用户发送到特定分区
            if key_str.startswith('VIP-'):
                return self.special_partition % num_partitions
            
            # 2. 普通 key 使用 hash
            if key_bytes is not None:
                return abs(mmh3.hash(key_bytes)) % num_partitions
        
        # 3. 无 key 使用轮询/粘性分区
        partitions = cluster.partitions_for_topic(topic)
        if not partitions:
            raise kafka_errors.UnproducibleMessage(topic, None)
        
        return next(self._partitioner) % len(partitions)
    
    def __call__(self):
        return self.partition


# Python 3 版本的简单分区器
def custom_partitioner(topic, key, key_bytes, value, value_bytes, cluster):
    """自定义分区函数"""
    num_partitions = cluster.partition_count_for_topic(topic)
    
    if key is None:
        # 无 key：使用 key_bytes 或随机选择
        return hash(value_bytes) % num_partitions if value_bytes else 0
    
    if key_bytes is not None:
        # 有 key：使用 hash
        return abs(mmh3.hash(key_bytes)) % num_partitions
    
    return abs(hash(key)) % num_partitions

  - 某些 key 消息量特别大（热点 key）
  - 导致某些分区压力过大

解决方案：
  1. 重新设计 key
     - 添加随机后缀：user-123-a, user-123-b
     - 缺点：失去顺序性保证

  2. 增加分区数
     - 分散到更多分区
     - 需要重新规划消费者

  3. 自定义分区器
     - 实现更复杂的分区逻辑
     - 基于消息内容动态分配

  4. 业务层拆分
     - 热点数据单独 Topic
     - 专用消费者处理

│                                                          │
│  Exactly Once (精确一次):                                │
│    • 消息不丢失、不重复                                  │
│    • 幂等性 + 事务 + 下游去重                            │

ISR (In-Sync Replicas):
  • 与 Leader 保持同步的副本集合
  • 由 Leader 维护

      ISR 虽然由 Leader 维护，但会同步存储到：

│                    极端场景模拟                              │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  初始状态：                                                │
│    • Controller: 节点1（唯一 Controller）                 │
│    • Broker A: Leader                                      │
│    • Broker B: Follower                                    │
│    • Broker C: Follower                                    │
│    • ISR = [A, B, C]                                      │
│                                                             │
│  极端故障：                                                │
│    • Controller 节点1 挂了                                │
│    • Leader A 同时挂了                                     │
│                                                             │
│  故障后：                                                  │
│    • ISR 信息保存在 ZooKeeper/KRaft，没丢！               │
│    • 但谁来执行选举？                                      │
│    • Controller 挂了，选举暂停！                          │
│                                                             │
│  恢复：                                                    │
│    • 等待 Controller 节点1 恢复                           │
│    • Controller 读取 ISR = [A, B, C]                      │
│    • 从 ISR 中选择新 Leader（B 或 C）                     │
│    • 选举完成                                              │
│                                                             │
└─────────────────────────────────────────────────────────────┘

判断标准:
  • replica.lag.time.max.ms (默认 30s)
  • Follower 在该时间内追上 Leader 即在 ISR 中

为什么需要 ISR:
  1. 平衡数据安全和可用性
     - 只需要 ISR 多数确认即可
     - 不需要所有副本确认
  
  2. 提供候选 Leader
     - Leader 故障时从 ISR 选举
     - 保证数据不丢失
  
  3. 防止数据丢失
     • min.insync.replicas 控制最小 ISR 数量
     • ISR 不足时拒绝写入

  • 分区数量变化
  • 心跳超时
  • 处理超时 (max.poll.interval.ms)

Rebalance 过程:
  1. Stop-the-world
     所有消费者停止消费，放弃分区所有权

  2. JoinGroup
     所有消费者向 Coordinator 发送 JoinGroup 请求
     Coordinator 选择一个 Leader

  3. 分区分配
     Leader 根据策略制定分配方案
     常见策略: Range, RoundRobin, Sticky

  4. SyncGroup
     Leader 发送分配方案给 Coordinator
     Coordinator 分发给所有消费者

  5. 恢复消费
     消费者从上次提交的偏移量继续消费

问题:
  • Rebalance 期间消费暂停
  • 频繁 Rebalance 影响性能

优化:
  • 使用 Sticky 分配策略
  • 合理配置心跳和会话超时
  • 避免频繁加入/退出消费者组

  1. 查找事务协调器
  2. 获取 Producer ID
  3. 开启事务

事务隔离级别:
  • read_uncommitted: 读取所有消息（默认）
