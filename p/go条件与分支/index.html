for range

Go语言中可以使用for range遍历数组、切片、字符串、map 及通道（channel）。 通过for range遍历的返回值有以下规律：

数组、切片、字符串返回索引和值。 map返回键和值。 通道（channel）只返回通道内的值。

Go1.22版本开始支持 for range 整数。类似 python 中的 range函数。

说明

我们使用了 for range 循环，来遍历了 data，得到了 key 和 value。

package main

import (
	"fmt"
	"strings"
)

func fordemo() {
	//老版本写法
	fmt.Println(strings.Repeat("-", 5) + "老版本 range 数字" + "-" + strings.Repeat("-", 5))
	for i := 0; i < 5; i++ {
		fmt.Println(i)
	}
	//1.22之后的写法
	fmt.Println(strings.Repeat("-", 5) + "1.22 range 数字" + "-" + strings.Repeat("-", 5))
	for i := range 5 {
		fmt.Println(i)
	}
	//数组、切片、字符串返回索引和值。
	fmt.Println(strings.Repeat("-", 5) + "遍历数组和切片" + "-" + strings.Repeat("-", 5))
	arr1 := []int{1, 2, 3, 4, 5}
	for i := range arr1 {
		fmt.Println(i, arr1[i])
	}
	//map返回key和value
	fmt.Println(strings.Repeat("-", 5) + "遍历map" + "-" + strings.Repeat("-", 5))
	m := map[string]int{
		"one":   1,
		"two":   2,
		"three": 3,
	}
	for k, v := range m {
		fmt.Println(k, v)
	}
	//channel返回值
	fmt.Println(strings.Repeat("-", 5) + "遍历channel" + "-" + strings.Repeat("-", 5))
	ch := make(chan int, 5)
	for i := 0; i < 5; i++ { //往channel中放入数据
		ch <- i
	}
	close(ch)           // ❗关键：通知消费端没有新数据了
	for v := range ch { //遍历channel中的数据
		fmt.Println(v)
	}
}

func main() {
	fordemo()
}

package main

import (
	"fmt"
	"time"
)

func switch_dema() {
	switch a := "1"; a {
	case "1":
		fmt.Println("1")
		fallthrough
	case "2":
		fmt.Println("2")
		fallthrough
	case "3":
		fmt.Println("3")
	case "4":
		fmt.Println("4")
	case "5":
		fmt.Println("5")
	default:
		fmt.Println("0")
	}
	// 输出1,2,3
	//无条件的switch,可以将一长串 if-then-else 写得更加清晰
	t := time.Now()
	switch {
	case t.Hour() < 12:
		fmt.Println("morning")
	case t.Hour() > 12 && t.Hour() < 17:
		fmt.Println("aftermoon")
	default:
		fmt.Println("evening")
	}
}

func main() {
	switch_dema()
}

package main

import (
	"fmt"
)

func break_demo() {
	//相当于是通过内层循环控制外层循环的 break
BREAKLEBAL:
	for i := range 10 {
		for j := range 5 {
			if j == 3 {
				break BREAKLEBAL
			}
			fmt.Println(i, "-", j)
		}
	}
}

func main() {
	break_demo()
}

continue

package main

import (
	"fmt"
)

func continue_demo() {
	//相当于是通过内层循环控制外层循环的 continue
CONTINUELEBAL:
	for i := range 3 {
		for j := range 5 {
			if j == 3 {
				continue CONTINUELEBAL
			}
			fmt.Println(i, "-", j)
		}
	}
}

func main() {
	continue_demo()
}
