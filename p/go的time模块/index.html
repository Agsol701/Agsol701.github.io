func main() {
	fmt.Println("time.Now():\t", time.Now())
	fmt.Println("time.Now().Year():\t", time.Now().Year())
	fmt.Println("time.Now().Month():\t", time.Now().Month())
	fmt.Println("time.Now().Day():\t", time.Now().Day())
	fmt.Println("time.Now().Hour():\t", time.Now().Hour())
	fmt.Println("time.Now().Minute():\t", time.Now().Minute())
	fmt.Println("time.Now().Second():\t", time.Now().Second())
	//时间戳
	timestamp := time.Now().Unix()
	fmt.Println("time.Now().Unix():", timestamp)
	//将时间戳转化为时间格式
	timeObj := time.Unix(timestamp, 0)
	fmt.Println("timeObj:", timeObj)
	//加时间，算过期时间
	fmt.Println("time after add:", timeObj.Add(time.Hour*24+time.Minute*30))
	//格式化时间 06年1月2日3时4分
	//24小时制
	fmt.Println(timeObj.Format("2006-01-02 15:04:05.000 Mon Jan"))
	//12小时制
	fmt.Println(timeObj.Format("2006-01-02 03:04:05.000 PM Mon Jan"))
	fmt.Println(timeObj.Format("03:04 2006/01/02"))
	//通过字符串时间转化为timeObj，字符串格式和format格式需要一致
	timeString := "2023-12-12 12:12:12"
	timeObj1, err := time.Parse("2006-01-02 03:04:05", timeString)
	if err != nil {
		fmt.Println("err:", err)
	}
	fmt.Println("timeOjb1:", timeObj1)
	//添加时区
	loc, err := time.LoadLocation("Asia/Shanghai")
	if err != nil {
		fmt.Println("err:", err)
	}
	timeObj2, err := time.ParseInLocation("2006-01-02 03:04:05", timeString, loc)
	if err != nil {
		fmt.Println("err:", err)
	}
	fmt.Println("timeOjb1:", timeObj2)
}
