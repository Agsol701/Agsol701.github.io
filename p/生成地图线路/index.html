from geopy.geocoders import Nominatim
import time
"""
pip install geopy
"""
def get_coordinates():
    geolocator = Nominatim(user_agent="my_map_route_app")
    
    locations = [
        "西宁市", "日月山", "黑马河镇", "青海湖", "茶卡盐湖", "德令哈市", 
        "可鲁克湖", "小柴旦湖", "东吉乃尔湖", "水上雅丹", 
        "西台吉乃尔湖", "冷湖镇", "茫崖翡翠湖", "花土沟镇", "艾肯泉", 
        "黑独山", "阿克塞石油小镇", "阳关景区", "敦煌市", "莫高窟", 
        "鸣沙山月牙泉", "瓜州县", "嘉峪关市", "张掖七彩丹霞", 
        "临泽县", "祁连县", "卓尔山", "岗什卡雪峰", "达坂山", "西宁市"
    ]
    
    # 手动补充一些 Nominatim 可能查不到或者不准确的特定景点
    # 这些坐标是大概位置，用于备用
    manual_coords = {
        "315国道": (37.7, 94.5), # 概略位置
        "火星一号公路": (38.0, 93.5), # 概略位置
        "大地之子": (40.5, 96.0), # 瓜州附近
        "黑独山": (38.8, 93.3),
        "阿克塞石油小镇": (39.5, 94.2),
        "水上雅丹": (37.45, 93.65),
        "东吉乃尔湖": (37.3, 93.9),
        "西台吉乃尔湖": (37.5, 93.5),
        "茫崖翡翠湖": (38.4, 90.7),
        "艾肯泉": (38.5, 90.6)
    }

    results = []
    
    for loc in locations:
        if loc in manual_coords:
            lat, lon = manual_coords[loc]
            print(f"Found manual: {loc} -> {lat}, {lon}")
            results.append((loc, lat, lon))
            continue

        try:
            # 尝试搜索
            location = geolocator.geocode(loc)
            if location:
                print(f"Found: {loc} -> {location.latitude}, {location.longitude}")
                results.append((loc, location.latitude, location.longitude))
            else:
                print(f"Not Found: {loc}")
                # 尝试加上 "青海" 或 "甘肃" 前缀重试
                if loc in ["日月山", "黑马河镇", "青海湖", "茶卡盐湖", "德令哈市", "可鲁克湖", "小柴旦湖", "冷湖镇", "花土沟镇", "卓尔山", "岗什卡雪峰", "达坂山"]:
                     query = "青海省 " + loc
                else:
                     query = "甘肃省 " + loc
                
                location = geolocator.geocode(query)
                if location:
                    print(f"Found (retry): {query} -> {location.latitude}, {location.longitude}")
                    results.append((loc, location.latitude, location.longitude))
                else:
                    # 如果还是找不到，检查是否在手动列表中
                    if loc in manual_coords:
                         lat, lon = manual_coords[loc]
                         results.append((loc, lat, lon))
                    else:
                         print(f"FAILED: {loc}")
                         results.append((loc, None, None))
            
            time.sleep(1) # Respect API rate limits
            
        except Exception as e:
            print(f"Error fetching {loc}: {e}")
            results.append((loc, None, None))
            
    print("\n--- Final List ---")
    for r in results:
        print(r)

if __name__ == "__main__":
    get_coordinates()

import folium
from folium import plugins
import os
"""
pip install folium
python others/map_trace_make/map_trace_make.py
"""
def create_map_route():
    route_name = "青甘大环线"
    # 1. 定义路线坐标点 (名称, 纬度, 经度)
    # 青甘大环线数据
    route_data = [
        ("西宁", 36.824463, 101.440811),
        ("日月山", 36.5645507, 100.9925056),
        ("黑马河镇", 36.7296869, 99.7773963),
        ("青海湖", 36.8927519, 100.0491803),
        ("茶卡盐湖", 36.6917272, 99.1136701),
        ("德令哈", 37.3690819, 97.36011),
        ("可鲁克湖", 37.2855682, 96.9039568),
        ("小柴旦湖", 37.4944818, 95.5072857),
        ("315国道(U型公路)", 37.6, 94.8),
        ("东吉乃尔湖", 37.3, 93.9),
        ("水上雅丹", 37.45, 93.65),
        ("西台吉乃尔湖", 37.5, 93.5),
        ("火星一号公路", 38.0, 93.5),
        ("冷湖镇", 38.7367371, 93.333492),
        ("茫崖翡翠湖", 38.4, 90.7),
        ("花土沟", 38.2521603, 90.8593199),
        ("艾肯泉", 38.5, 90.6),
        ("黑独山", 38.8, 93.3),
        ("阿克塞石油小镇", 39.5, 94.2),
        ("阳关", 39.9354508, 94.0535644),
        ("敦煌", 40.1411004, 94.6607593),
        ("莫高窟", 40.0373163, 94.8042114),
        ("鸣沙山月牙泉", 40.0876404, 94.6753955),
        ("瓜州", 40.5187181, 95.7810399),
        ("大地之子", 40.5, 96.0),
        ("嘉峪关", 39.7722294, 98.2885388),
        ("张掖七彩丹霞", 38.9712611, 100.0617058),
        ("临泽县", 39.347318, 100.188022),
        ("祁连县", 38.2068099, 99.8413546),
        ("卓尔山", 38.17, 100.02),
        ("岗什卡雪峰", 37.69, 101.45),
        ("达坂山", 37.33, 101.40),
        ("西宁", 36.824463, 101.440811)
    ]

    if not route_data:
        print("没有路线数据")
        return

    # 提取坐标列表用于绘制线条
    route_coords = [(lat, lon) for name, lat, lon in route_data]

    # 2. 创建地图对象
    # 自动计算中心点
    avg_lat = sum(p[0] for p in route_coords) / len(route_coords)
    avg_lon = sum(p[1] for p in route_coords) / len(route_coords)
    
    # zoom_start 设为 6 或 7 以便看到整个西北环线
    m = folium.Map(location=[avg_lat, avg_lon], zoom_start=6)

    # 配置项
    show_arrows = False  # 是否显示方向箭头
    
    marker_config = {
        "radius": 4,          # 标记点半径大小
        "color": "red",       # 标记点边框颜色
        "fill_color": "white",# 标记点填充颜色
        "fill_opacity": 1     # 填充透明度
    }

    # 3. 添加路线
    folium.PolyLine(
        locations=route_coords,
        color="blue",
        weight=4,
        opacity=0.7,
        tooltip="青甘大环线"
    ).add_to(m)

    # 添加每一段中间的红色箭头 (如果开启)
    if show_arrows:
        for i in range(len(route_coords) - 1):
            # 创建每一段的辅助线（透明），用于定位箭头
            segment = folium.PolyLine(
                locations=[route_coords[i], route_coords[i+1]],
                opacity=0,
                weight=0
            ).add_to(m)
            
            # 在辅助线中间添加红色箭头
            plugins.PolyLineTextPath(
                segment,
                '►',
                repeat=False,
                center=True,      # 居中显示
                offset=7,         # 偏移量，使箭头对齐线条
                attributes={'fill': 'red', 'font-size': '20', 'font-weight': 'bold'}
            ).add_to(m)

    # 4. 添加标记
    for i, (name, lat, lon) in enumerate(route_data):
        # 起点和终点特殊标记
        if i == 0:
            folium.Marker(
                location=(lat, lon),
                popup=f"起点: {name}",
                icon=folium.Icon(color="green", icon="play")
            ).add_to(m)
        elif i == len(route_data) - 1:
            folium.Marker(
                location=(lat, lon),
                popup=f"终点: {name}",
                icon=folium.Icon(color="red", icon="stop")
            ).add_to(m)
        else:
            # 中间点使用小圆点或小图标
            # 为了避免地图太乱，可以使用 CircleMarker 并带有 tooltip
            folium.CircleMarker(
                location=(lat, lon),
                radius=marker_config["radius"],
                color=marker_config["color"],
                fill=True,
                fill_color=marker_config["fill_color"],
                fill_opacity=marker_config["fill_opacity"],
                popup=name,
                tooltip=f"{i}. {name}"  # 鼠标悬停显示名称和序号
            ).add_to(m)
            
            # 如果想让重要地点的名字直接显示在地图上，可以使用 DivIcon (可选，这里先只用 Tooltip)

    # 5. 保存地图为 HTML 文件
    # 获取当前脚本所在目录
    current_dir = os.path.dirname(os.path.abspath(__file__))
    output_file = os.path.join(current_dir, f"{route_name}_map_route.html")
    
    m.save(output_file)
    print(f"地图已生成并保存为: {output_file}")
    print("请刷新浏览器查看最新的青甘大环线地图。")

if __name__ == "__main__":
    create_map_route()
